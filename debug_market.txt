   117      void _listenStocksRealtime() {
   118        _stocksChannel = Supabase.instance.client
   119            .channel('stocks-live')
   120            .onPostgresChanges(
   121              event: PostgresChangeEvent.all,
   122              schema: 'public',
   123              table: 'stocks',
   124              callback: (payload) {
   125                final rec = payload.newRecord;
   126                if (rec.isEmpty) return;
   127                if (rec['is_active'] == false) return;
   128    
   129                final updated = StockModel.fromMap(rec);
   130    
   131                final prev = _lastPrice[updated.symbol];
   132                if (prev != null) {
   133                  final dir = updated.price > prev
   134                      ? 1
   135                      : (updated.price < prev ? -1 : 0);
   136                  if (dir != 0) _triggerFlash(updated.symbol, dir);
   137                }
   138                _lastPrice[updated.symbol] = updated.price;
   139    
   140                final idx = _stocks.indexWhere((s) => s.symbol == updated.symbol);
   141                setState(() {
   142                  if (idx == -1) {
   143                    _stocks = [..._stocks, updated]
   144                      ..sort((a, b) => a.symbol.compareTo(b.symbol));
   145                  } else {
   146                    final copy = [..._stocks];
   147                    copy[idx] = updated;
   148                    _stocks = copy;
   149                  }
   150                });
   151              },
   152            )
   153            .subscribe();
   154      }

   122

   124          final changePercent =

   126              ? computedChange
   127              : parsedChange;
   128    
   129          final volume = _int(cols[4].text);
   130          final turnover = cols.length >= 6 ? _num(cols[5].text) : 0.0;
   131

   133            'symbol': symbol,

   135            'price': closePrice,

   138            'turnover_mwk': turnover,
   139            'updated_at': now,
   140          });
   141        }
   142    
   143        if (updates.isEmpty) {
   144          throw 'No matching symbols updated. (DB symbols may not match MSE page symbols)';
   145        }
   146    
   147        await _db.from('stocks').upsert(updates, onConflict: 'symbol');
   148        await _insertHistory(updates, now);
   149    
   150        return updates.length;
   151      }
