     9	  Future<String> createMarketRequestOrder({
    10	    required StockModel stock,
    11	    required BrokerModel broker,
    12	    required String side,
    13	    required int quantity,
    14	    String? investorNote,
    15	  }) async {
    16	    final user = _client.auth.currentUser;
    17	    if (user == null) throw StateError('User not logged in');
    18	
    19	    final priceAtSubmission = stock.price;
    20	    final feeRate = broker.feeRate;
    21	
    22	    final subtotal = quantity * priceAtSubmission;
    23	    final feeAmount = subtotal * feeRate;
    24	    final totalEstimate = subtotal + feeAmount;
    25	
    26	    final inserted = await _client
    27	        .from('trade_orders')
    28	        .insert({
    29	          'user_id': user.id,
    30	          'broker_id': broker.id,
    31	          'stock_symbol': stock.symbol,
    32	          'side': side,
    33	          'quantity': quantity,
    34	          'price_at_submission': priceAtSubmission,
    35	          'fee_rate': feeRate,
    36	          'fee_amount': feeAmount,
    37	          'total_estimate': totalEstimate,
    38	          'status': 'submitted',
    39	          'investor_note': investorNote,
    40	          'deleted_at': null,
    41	        })
    42	        .select('id')
    43	        .single();
    44	
    45	    return inserted['id'] as String;
    46	  }

    48	  Future<List<TradeOrderModel>> getMyOrders({int limit = 50}) async {
    49	    final user = _client.auth.currentUser;
    50	    if (user == null) throw StateError('User not logged in');
    51	
    52	    final resp = await _client
    53	        .from('trade_orders')
    54	        .select(
    55	          'id, stock_symbol, side, quantity, status, broker_id, total_estimate, created_at, updated_at',
    56	        )
    57	        .eq('user_id', user.id)
    58	        .isFilter('deleted_at', null)
    59	        .order('created_at', ascending: false)
    60	        .limit(limit);
    61	
    62	    final orders = (resp as List)
    63	        .map((e) => TradeOrderModel.fromMap(e as Map<String, dynamic>))
    64	        .toList();
    65	
    66	    if (orders.isEmpty) return [];
    67	
    68	    final brokerIds = orders.map((o) => o.brokerId).toSet().toList();
    69	    final brokersResp = await _client
    70	        .from('brokers')
    71	        .select('id,name')
    72	        .inFilter('id', brokerIds);
    73	
    74	    final brokers = (brokersResp as List).cast<Map<String, dynamic>>();
    75	    final nameById = {
    76	      for (final b in brokers) b['id'] as String: b['name'] as String,
    77	    };
    78	
    79	    return orders
    80	        .map((o) => o.copyWith(brokerName: nameById[o.brokerId]))
    81	        .toList();
    82	  }
    83	
    84	  Future<void> softDeleteMyOrder(String orderId) async {
    85	    final user = _client.auth.currentUser;
    86	    if (user == null) throw StateError('User not logged in');
    87	
    88	    final row = await _client
    89	        .from('trade_orders')
    90	        .select('status')
    91	        .eq('id', orderId)
    92	        .eq('user_id', user.id)
    93	        .maybeSingle();
    94	
    95	    final status = (row?['status'] ?? 'submitted').toString().toLowerCase();
    96	    if (status == 'executed' || status == 'settled') {
    97	      throw StateError('Cannot delete an $status order');
    98	    }
    99	
   100	    await _client
   101	        .from('trade_orders')
   102	        .update({'deleted_at': DateTime.now().toUtc().toIso8601String()})
   103	        .eq('id', orderId)
   104	        .eq('user_id', user.id);
   105	  }
