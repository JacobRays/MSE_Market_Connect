   136      void _listenRealtime() {

   140        _ordersChannel = Supabase.instance.client
   141            .channel('orders-live-${user.id}')
   142            .onPostgresChanges(
   143              event: PostgresChangeEvent.all,
   144              schema: 'public',
   145              table: 'trade_orders',
   146              callback: (payload) {
   147                final rec = payload.newRecord.isNotEmpty ? payload.newRecord : payload.oldRecord;
   148                if (rec.isEmpty) return;

    91        if (hasMore) list = list.sublist(0, limit);

    94        if (models.isEmpty) return OrdersPage(items: const [], hasMore: false, nextOffset: offset);
    95    
    96        final brokerIds = models.map((o) => o.brokerId).toSet().toList();

   100            .inFilter('id', brokerIds);

   103        final nameById = {for (final b in brokers) b['id'] as String: b['name'] as String};
   104    
   105        final enriched = models.map((o) => o.copyWith(brokerName: nameById[o.brokerId])).toList();
   106        return OrdersPage(items: enriched, hasMore: hasMore, nextOffset: offset + enriched.length);
   107      }
   108    
   109    
   110        q = q.order('created_at', ascending: ascending);
   111    
   112        // Ask for limit+1 (inclusive range end) to detect "hasMore"
   113        final rows = await q.range(offset, offset + limit);
   114

   117        if (hasMore,) list = list.sublist(0, limit);
   118

   121        if (models.isEmpty) return OrdersPage(items: const [], hasMore: false, nextOffset: offset);
   122    
   123        final brokerIds = models.map((o) => o.brokerId).toSet().toList();
   124        final brokersResp = await _client
